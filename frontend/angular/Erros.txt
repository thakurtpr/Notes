Error :=   @Input() childProperty: string ;


    This error occurs because TypeScript has a feature called "definite assignment assertions" 
    which checks if a property is assigned a value before it's used.


Solution :=

Method 1: Initialize the property

    import { Component, Input } from '@angular/core';

    @Component({
    selector: 'app-child',
    template: `
        <p>Received from Parent: {{ childProperty }}</p>
    `
    })
    export class ChildComponent {
      @Input() childProperty: string = '';
    }

Method 2: Use the ? optional property operator

import { Component, Input } from '@angular/core';

    @Component({
    selector: 'app-child',
    template: `
        <p>Received from Parent: {{ childProperty }}</p>
    `
    })
    export class ChildComponent {
        @Input() childProperty?: string;
    }

Method 3: Use the ! non-null assertion operator

    import { Component, Input } from '@angular/core';

    @Component({
    selector: 'app-child',
    template: `
        <p>Received from Parent: {{ childProperty }}</p>
    `
    })
    export class ChildComponent {
        @Input() childProperty!: string;
    }

Method 4: Add a constructor with a default value

    import { Component, Input } from '@angular/core';

    @Component({
    selector: 'app-child',
    template: `
        <p>Received from Parent: {{ childProperty }}</p>
    `
    })
    export class ChildComponent {
    @Input() childProperty: string;

    constructor() {
        this.childProperty = '';
    }
    }
